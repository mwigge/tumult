# The raw style values that Krönika keeps, one `path kind value` per line,
# with paths from web/. A value used twice in a file is on two lines. The
# style lint (npm run lint:styles) fails on a raw value that is not here,
# and on a line that matches no raw value. Each value here is one that a
# token cannot hold, and the comment above it says why. Do not add lines:
# use a token.

# The shadow of a drawer and the shade over the page behind it: black at
# an opacity, the same in both themes, so they are no colour role.
src/lib/components/Drawer.svelte colour rgb(0 0 0 / 25%)
src/lib/components/Drawer.svelte colour rgba(0, 0, 0, 0.45)

# Mono text looks larger than the body font at the same size, so its size
# is a share of the text around it, not a step of the scale.
src/lib/theme.css font-size 0.92em

# A layout trick: the spare pixel at the end of the navigation keeps its
# last entry whole while the scroll position is not a whole pixel.
src/lib/theme.css spacing 1px

# A layout trick: the summary moves up to the title of its panel.
src/routes/dr/+page.svelte spacing -4px

# A layout trick: the hint moves up to the field it explains.
src/routes/login/+page.svelte spacing -4px

# A layout trick: the filter chips move up to the head of the page.
src/routes/logs/+page.svelte spacing -6px
src/routes/traces/+page.svelte spacing -6px

# The ring of a node in the key of the Star Map is 2 px when it is ok,
# 3 px when it warns and 4 px when it fails, so that its width shows the
# state without its colour.
src/routes/starmap/+page.svelte border-width 3px
